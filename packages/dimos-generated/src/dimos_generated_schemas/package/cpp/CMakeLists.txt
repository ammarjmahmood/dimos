cmake_minimum_required(VERSION 3.20)
project(dimos_generated VERSION 0.1.0 LANGUAGES CXX)
find_package(fastcdr 2.4.0 EXACT REQUIRED)
include(GNUInstallDirs)
add_library(dimos_generated_messages INTERFACE)
set_target_properties(dimos_generated_messages PROPERTIES EXPORT_NAME messages)
target_compile_features(dimos_generated_messages INTERFACE cxx_std_17)
target_include_directories(dimos_generated_messages INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:include>)
target_link_libraries(dimos_generated_messages INTERFACE fastcdr)
install(TARGETS dimos_generated_messages EXPORT dimos_generatedTargets)
install(FILES messages.hpp dimos_cdr.hpp DESTINATION include/dimos_generated)
install(EXPORT dimos_generatedTargets NAMESPACE dimos_generated:: DESTINATION lib/cmake/dimos_generated)
install(FILES dimos_generatedConfig.cmake DESTINATION lib/cmake/dimos_generated)
install(DIRECTORY ../schemas DESTINATION share/dimos_generated)
